47/// unwrap the JsonObject under [`serde_json::Value`]
48///
49/// # Panic

52    debug_assert!(value.is_object());
53    match value {
54        serde_json::Value::Object(map) => map,

169impl ProtocolVersion {
170    pub const V_2026_07_28: Self = Self(Cow::Borrowed("2026-07-28"));
171    pub const V_2025_11_25: Self = Self(Cow::Borrowed("2025-11-25"));
172    pub const V_2025_06_18: Self = Self(Cow::Borrowed("2025-06-18"));
173    pub const V_2025_03_26: Self = Self(Cow::Borrowed("2025-03-26"));
174    pub const V_2024_11_05: Self = Self(Cow::Borrowed("2024-11-05"));
175    pub const LATEST: Self = Self::V_2025_11_25;
176
177    /// First protocol version that requires SEP-2243 standard HTTP headers.
178    pub const STANDARD_HEADERS: Self = Self::V_2026_07_28;
179
180    /// All protocol versions known to this SDK, oldest first.
181    pub const KNOWN_VERSIONS: &[Self] = &[
182        Self::V_2024_11_05,
183        Self::V_2025_03_26,
184        Self::V_2025_06_18,
185        Self::V_2025_11_25,
186        Self::V_2026_07_28,
187    ];
188
189    /// Returns the string representation of this protocol version.
190    pub fn as_str(&self) -> &str {
191        &self.0
192    }
193
194    /// The known versions up to and including `max`, oldest first.
195    ///
196    /// Servers that implement every revision up to some ceiling can return
197    /// this from `supported_protocol_versions` instead of filtering
198    /// [`Self::KNOWN_VERSIONS`] by hand. `max` itself need not be a known
199    /// version; the result is empty when it predates all of them.
200    ///
201    /// The result borrows from [`Self::KNOWN_VERSIONS`], so call it directly
202    /// in the method body — it needs no `static` and no `LazyLock`:
203    ///
204    /// ```rust,ignore
205    /// const MAX_SUPPORTED: ProtocolVersion = ProtocolVersion::V_2025_11_25;
206    ///
207    /// fn supported_protocol_versions(&self) -> Cow<'static, [ProtocolVersion]> {
208    ///     Cow::Borrowed(ProtocolVersion::known_up_to(&MAX_SUPPORTED))
209    /// }
210    /// ```
211    ///
212    /// ```
213    /// # use rmcp::model::ProtocolVersion;
214    /// assert_eq!(
215    ///     ProtocolVersion::known_up_to(&ProtocolVersion::V_2025_06_18),
216    ///     &[
217    ///         ProtocolVersion::V_2024_11_05,
218    ///         ProtocolVersion::V_2025_03_26,
219    ///         ProtocolVersion::V_2025_06_18,
220    ///     ],
221    /// );
222    /// ```
223    pub fn known_up_to(max: &Self) -> &'static [Self] {
224        let count = Self::KNOWN_VERSIONS
225            .iter()
226            .take_while(|version| version.as_str() <= max.as_str())
227            .count();
228        &Self::KNOWN_VERSIONS[..count]
229    }
230}

260/// A flexible identifier type that can be either a number or a string.
261///
262/// This is commonly used for request IDs and other identifiers in JSON-RPC
263/// where the specification allows both numeric and string values.
264#[derive(Debug, Clone, Eq, PartialEq, Hash)]
265#[expect(clippy::exhaustive_enums, reason = "intentionally exhaustive")]
266pub enum NumberOrString {
267    /// A numeric identifier
268    Number(i64),
269    /// A string identifier
270    String(Arc<str>),
271}
272
273impl NumberOrString {
274    pub fn into_json_value(self) -> Value {
275        match self {
276            NumberOrString::Number(n) => Value::Number(serde_json::Number::from(n)),
277            NumberOrString::String(s) => Value::String(s.to_string()),
278        }
279    }
280
281    pub(crate) fn numeric_string_value(&self) -> Option<i64> {
282        match self {
283            Self::String(id) => id.parse().ok(),
284            Self::Number(_) => None,
285        }
286    }
287
288    pub(crate) fn matches_response_id(&self, response_id: &Self) -> bool {
289        self == response_id
290            || matches!(
291                self,
292                Self::Number(request_id)
293                    if response_id.numeric_string_value() == Some(*request_id)
294            )
295    }
296}

319impl<'de> Deserialize<'de> for NumberOrString {
320    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
321    where
322        D: serde::Deserializer<'de>,
323    {
324        let value: Value = Deserialize::deserialize(deserializer)?;
325        match value {
326            Value::Number(n) => {
327                if let Some(i) = n.as_i64() {
328                    Ok(NumberOrString::Number(i))
329                } else if let Some(u) = n.as_u64() {
330                    // Handle large unsigned numbers that fit in i64
331                    if u <= i64::MAX as u64 {
332                        Ok(NumberOrString::Number(u as i64))
333                    } else {
334                        Err(serde::de::Error::custom("Number too large for i64"))
335                    }
336                } else {
337                    Err(serde::de::Error::custom("Expected an integer"))
338                }
339            }
340            Value::String(s) => Ok(NumberOrString::String(s.into())),
341            _ => Err(serde::de::Error::custom("Expect number or string")),
342        }
343    }
344}

705/// Represents any JSON-RPC message that can be sent or received.
706///
707/// This enum covers all possible message types in the JSON-RPC protocol:
708/// individual requests/responses, notifications, and errors.
709/// It serves as the top-level message container for MCP communication.
710#[derive(Debug, Serialize, Deserialize, Clone, PartialEq)]
711#[serde(untagged)]
712#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
713#[expect(clippy::exhaustive_enums, reason = "intentionally exhaustive")]
714pub enum JsonRpcMessage<Req = Request, Resp = DefaultResponse, Noti = Notification> {
715    /// A single request expecting a response
716    Request(JsonRpcRequest<Req>),
717    /// A response to a previous request
718    Response(JsonRpcResponse<Resp>),
719    /// A one-way notification (no response expected)
720    Notification(JsonRpcNotification<Noti>),
721    /// An error response
722    Error(JsonRpcError),
723}
724
725impl<Req, Resp, Not> JsonRpcMessage<Req, Resp, Not> {
726    #[inline]
727    pub const fn request(request: Req, id: RequestId) -> Self {
728        JsonRpcMessage::Request(JsonRpcRequest {
729            jsonrpc: JsonRpcVersion2_0,
730            id,
731            request,
732        })
733    }
734    #[inline]
735    pub const fn response(response: Resp, id: RequestId) -> Self {
736        JsonRpcMessage::Response(JsonRpcResponse {
737            jsonrpc: JsonRpcVersion2_0,
738            id,
739            result: response,
740        })
741    }
742    #[inline]
743    pub const fn error(error: ErrorData, id: Option<RequestId>) -> Self {
744        JsonRpcMessage::Error(JsonRpcError {
745            jsonrpc: JsonRpcVersion2_0,
746            id,
747            error,
748        })
749    }
750    #[inline]
751    pub const fn notification(notification: Not) -> Self {
752        JsonRpcMessage::Notification(JsonRpcNotification {
753            jsonrpc: JsonRpcVersion2_0,
754            notification,
755        })
756    }
757    pub fn into_request(self) -> Option<(Req, RequestId)> {
758        match self {
759            JsonRpcMessage::Request(r) => Some((r.request, r.id)),
760            _ => None,
761        }
762    }
763    pub fn into_response(self) -> Option<(Resp, RequestId)> {
764        match self {
765            JsonRpcMessage::Response(r) => Some((r.result, r.id)),
766            _ => None,
767        }
768    }
769    pub fn into_notification(self) -> Option<Not> {
770        match self {
771            JsonRpcMessage::Notification(n) => Some(n.notification),
772            _ => None,
773        }
774    }
775    pub fn into_error(self) -> Option<(ErrorData, Option<RequestId>)> {
776        match self {
777            JsonRpcMessage::Error(e) => Some((e.error, e.id)),
778            _ => None,
779        }
780    }
781    pub fn into_result(self) -> Option<(Result<Resp, ErrorData>, Option<RequestId>)> {
782        match self {
783            JsonRpcMessage::Response(r) => Some((Ok(r.result), Some(r.id))),
784            JsonRpcMessage::Error(e) => Some((Err(e.error), e.id)),
785
786            _ => None,
787        }
788    }

810/// determine how to parse the response.
811///
812/// The spec defines this as an open string (`"complete" | "input_required" | string`),
813/// so unknown values are preserved rather than rejected. Servers implementing this
814/// protocol version MUST include `resultType` in every result. For backward
815/// compatibility, clients MUST treat an absent field as `"complete"`.
816///
817/// Ordinary results model the field as `Option<ResultType>`: `None` means the
818/// field is absent on the wire. Constructors default to `Some(COMPLETE)`, and
819/// the server handler strips the `"complete"` discriminator before responding
820/// to peers that negotiated a protocol version older than `2026-07-28`, so
821/// legacy sessions keep their historical wire shape (see
822/// [`ServerResult::strip_result_type_for_legacy_peer`]).

829    pub const INPUT_REQUIRED: Self = Self(Cow::Borrowed("input_required"));
830    /// SEP-2663 Tasks extension: the result is a task handle ([`CreateTaskResult`]).
831    pub const TASK: Self = Self(Cow::Borrowed("task"));
832
833    pub fn as_str(&self) -> &str {
834        &self.0
835    }
836
837    /// Returns `true` if this is `"input_required"`.
838    pub fn is_input_required(&self) -> bool {

840    }

842    /// Returns `true` if this is `"complete"`.

844        self.0 == "complete"
845    }
846

850    }

1599/// Normalize a `ttlMs` value during deserialization.
1600///
1601/// Per SEP-2549, `ttlMs` MUST be `>= 0`; if a server returns a negative value,
1602/// clients SHOULD treat it as `0` (immediately stale). This tolerates that case
1603/// rather than erroring, while still accepting an absent field as `None`.
1604fn deserialize_ttl_ms<'de, D>(deserializer: D) -> Result<Option<u64>, D::Error>
1605where
1606    D: serde::Deserializer<'de>,
1607{
1608    let value = Option::<i64>::deserialize(deserializer)?;
1609    Ok(value.map(|ttl_ms| ttl_ms.max(0) as u64))
1610}

1956    #[cfg_attr(feature = "schemars", schemars(with = "bool"))]
1957    pub tools_list_changed: Option<bool>,

1959    #[cfg_attr(feature = "schemars", schemars(with = "bool"))]

1961    #[serde(default, skip_serializing_if = "Option::is_none")]
1962    #[cfg_attr(feature = "schemars", schemars(with = "bool"))]
1963    pub resources_list_changed: Option<bool>,
1964    #[serde(default, skip_serializing_if = "Option::is_none")]
1965    #[cfg_attr(feature = "schemars", schemars(with = "Vec<String>"))]
1966    pub resource_subscriptions: Option<Vec<String>>,
1967}

1969impl SubscriptionFilter {
1970    /// Create an empty filter that opts in to no notifications.
1971    pub fn new() -> Self {
1972        Self::default()
1973    }
1974
1975    /// Create a builder for a subscription filter.
1976    pub fn builder() -> SubscriptionFilterBuilder {
1977        SubscriptionFilterBuilder::default()
1978    }
1979
1980    /// Return the subset present in both filters.
1981    pub fn intersection(&self, other: &Self) -> Self {
1982        let resource_subscriptions = self
1983            .resource_subscriptions
1984            .as_ref()
1985            .and_then(|requested| {
1986                other.resource_subscriptions.as_ref().map(|accepted| {
1987                    requested
1988                        .iter()
1989                        .filter(|uri| accepted.contains(uri))
1990                        .cloned()
1991                        .collect()
1992                })
1993            })
1994            .filter(|uris: &Vec<String>| !uris.is_empty());
1995        Self {
1996            tools_list_changed: (self.tools_list_changed == Some(true)
1997                && other.tools_list_changed == Some(true))
1998            .then_some(true),
1999            prompts_list_changed: (self.prompts_list_changed == Some(true)
2000                && other.prompts_list_changed == Some(true))
2001            .then_some(true),
2002            resources_list_changed: (self.resources_list_changed == Some(true)
2003                && other.resources_list_changed == Some(true))
2004            .then_some(true),
2005            resource_subscriptions,
2006        }
2007    }
2008
2009    /// Return whether this filter accepts only notifications requested by `other`.
2010    pub fn is_subset_of(&self, other: &Self) -> bool {
2011        let booleans_are_subset = [
2012            (self.tools_list_changed, other.tools_list_changed),
2013            (self.prompts_list_changed, other.prompts_list_changed),
2014            (self.resources_list_changed, other.resources_list_changed),
2015        ]
2016        .into_iter()
2017        .all(|(accepted, requested)| accepted != Some(true) || requested == Some(true));
2018        let resources_are_subset = self.resource_subscriptions.as_ref().is_none_or(|accepted| {
2019            accepted.iter().all(|uri| {
2020                other
2021                    .resource_subscriptions
2022                    .as_ref()
2023                    .is_some_and(|requested| requested.contains(uri))
2024            })
2025        });
2026        booleans_are_subset && resources_are_subset
2027    }
2028
2029    /// Return the requested notification types advertised by server capabilities.
2030    pub fn supported_by(&self, capabilities: &ServerCapabilities) -> Self {
2031        Self {
2032            tools_list_changed: (self.tools_list_changed == Some(true)
2033                && capabilities
2034                    .tools
2035                    .as_ref()
2036                    .is_some_and(|tools| tools.list_changed == Some(true)))
2037            .then_some(true),
2038            prompts_list_changed: (self.prompts_list_changed == Some(true)
2039                && capabilities
2040                    .prompts
2041                    .as_ref()
2042                    .is_some_and(|prompts| prompts.list_changed == Some(true)))
2043            .then_some(true),
2044            resources_list_changed: (self.resources_list_changed == Some(true)
2045                && capabilities
2046                    .resources
2047                    .as_ref()
2048                    .is_some_and(|resources| resources.list_changed == Some(true)))
2049            .then_some(true),
2050            resource_subscriptions: capabilities
2051                .resources
2052                .as_ref()
2053                .is_some_and(|resources| resources.subscribe == Some(true))
2054                .then(|| self.resource_subscriptions.clone())
2055                .flatten(),
2056        }
2057    }
2058}

3011    /// Validate the sampling request parameters per SEP-1577 spec requirements.
3012    ///
3013    /// Checks:
3014    /// - ToolUse content is only allowed in assistant messages
3015    /// - ToolResult content is only allowed in user messages
3016    /// - Messages with tool result content MUST NOT contain other content types
3017    /// - Every assistant ToolUse must be balanced with a corresponding user ToolResult
3018    pub fn validate(&self) -> Result<(), String> {
3019        for msg in &self.messages {
3020            for content in msg.content.iter() {
3021                // ToolUse only in assistant messages, ToolResult only in user messages
3022                match content {
3023                    SamplingMessageContentBlock::ToolUse(_) if msg.role != Role::Assistant => {
3024                        return Err("ToolUse content is only allowed in assistant messages".into());
3025                    }
3026                    SamplingMessageContentBlock::ToolResult(_) if msg.role != Role::User => {
3027                        return Err("ToolResult content is only allowed in user messages".into());
3028                    }
3029                    _ => {}
3030                }
3031            }
3032
3033            // Tool result messages MUST NOT contain other content types
3034            let contents: Vec<_> = msg.content.iter().collect();
3035            let has_tool_result = contents
3036                .iter()
3037                .any(|c| matches!(c, SamplingMessageContentBlock::ToolResult(_)));
3038            if has_tool_result
3039                && contents
3040                    .iter()
3041                    .any(|c| !matches!(c, SamplingMessageContentBlock::ToolResult(_)))
3042            {
3043                return Err(
3044                    "SamplingMessage with tool result content MUST NOT contain other content types"
3045                        .into(),
3046                );
3047            }
3048        }
3049
3050        // Every assistant ToolUse must be balanced with a user ToolResult
3051        self.validate_tool_use_result_balance()?;
3052
3053        Ok(())
3054    }
3055
3056    fn validate_tool_use_result_balance(&self) -> Result<(), String> {
3057        let mut pending_tool_use_ids: Vec<String> = Vec::new();
3058        for msg in &self.messages {
3059            if msg.role == Role::Assistant {
3060                for content in msg.content.iter() {
3061                    if let SamplingMessageContentBlock::ToolUse(tu) = content {
3062                        pending_tool_use_ids.push(tu.id.clone());
3063                    }
3064                }
3065            } else if msg.role == Role::User {
3066                for content in msg.content.iter() {
3067                    if let SamplingMessageContentBlock::ToolResult(tr) = content {
3068                        if !pending_tool_use_ids.contains(&tr.tool_use_id) {
3069                            return Err(format!(
3070                                "ToolResult with toolUseId '{}' has no matching ToolUse",
3071                                tr.tool_use_id
3072                            ));
3073                        }
3074                        pending_tool_use_ids.retain(|id| id != &tr.tool_use_id);
3075                    }
3076                }
3077            }
3078        }
3079        if !pending_tool_use_ids.is_empty() {
3080            return Err(format!(
3081                "ToolUse with id(s) {:?} not balanced with ToolResult",
3082                pending_tool_use_ids
3083            ));
3084        }
3085        Ok(())
3086    }

3933    /// MCP distinguishes two failure modes for a `call_tool` invocation, and
3934    /// the right one to use depends on **whose problem it is**:

3937    ///   The request was valid and routed to your tool, but executing the

3944    /// - **Protocol error** — `Err(ErrorData)` with a JSON-RPC code.
3945    ///   The server cannot route the request at all, or an infrastructure

4634        let result_type = match self {
4635            ServerResult::CompleteResult(r) => &mut r.result_type,
4636            ServerResult::GetPromptResult(r) => &mut r.result_type,
4637            ServerResult::ListPromptsResult(r) => &mut r.result_type,
4638            ServerResult::ListResourcesResult(r) => &mut r.result_type,
4639            ServerResult::ListResourceTemplatesResult(r) => &mut r.result_type,

4641            ServerResult::ListToolsResult(r) => &mut r.result_type,
4642            ServerResult::CallToolResult(r) => &mut r.result_type,
