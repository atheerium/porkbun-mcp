1//! Multi Round-Trip Request (MRTR) types for SEP-2322.
2//!
3//! Provides [`InputRequiredResult`], [`InputRequests`], and [`InputResponses`]
4//! for the stateless multi round-trip request pattern defined in the MCP spec.
5//! [`ResultType`] lives in the parent [`super`] module alongside other base result types.
6//!
7//! # Overview
8//!
9//! A server may respond to `tools/call`, `prompts/get`, or `resources/read` with an
10//! [`InputRequiredResult`] instead of the normal result. The client fulfills the
11//! [`InputRequests`], then retries the original request with [`InputResponses`] and
12//! the echoed `requestState`.
13//!
14//! # Using MRTR
15//!
16//! **Server:** return an [`InputRequiredResult`] from a tool/prompt/resource
17//! handler via the matching outcome enum ([`CallToolResponse`],
18//! [`GetPromptResponse`], [`ReadResourceResponse`]). The SDK only lets an
19//! `InputRequiredResult` reach a peer that negotiated protocol version
20//! `2026-07-28` or newer; older peers get a protocol error instead.
21//!
22//! **Client:** the high-level `RunningService` helpers — `call_tool`,
23//! `get_prompt`, and `read_resource` — automatically fulfil each
24//! [`InputRequest`] through the local `ClientHandler` and retry, up to
25//! [`DEFAULT_MRTR_MAX_ROUNDS`]. Use the `*_once` variants (e.g.
26//! `call_tool_once`) to receive an [`InputRequiredResult`] directly and drive
27//! the rounds yourself.
28//!
29//! # `requestState` is untrusted
30//!
31//! The client echoes `requestState` back verbatim, so a stateless server that
32//! stores meaningful data in it MUST verify integrity before trusting the echoed
33//! value. Enable the `request-state` feature and use `RequestStateCodec` to seal
34//! and open it, or keep the state server-side and use `requestState` only as an
35//! opaque handle.
36//!
37//! A complete runnable walkthrough lives in the `servers_mrtr` example.
38

49/// Default maximum number of MRTR rounds a high-level client call will drive.
50///
51/// This matches the default used by other Tier 1 SDKs and prevents a
52/// misbehaving peer from keeping a request alive indefinitely.
53pub const DEFAULT_MRTR_MAX_ROUNDS: usize = 10;

55/// A server-initiated request that can appear inside [`InputRequests`].
56///
57/// Per the MCP spec, only `CreateMessageRequest` (sampling),
58/// `ElicitRequest` (elicitation), and `ListRootsRequest` (roots)
59/// are allowed. This is modeled as an untagged enum rather than a
60/// `ServerRequest` alias to prevent `PingRequest` or `CustomRequest` from
61/// being included.
62#[derive(Debug, Clone, Serialize, Deserialize)]
63#[serde(untagged)]
64#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
65#[non_exhaustive]
66pub enum InputRequest {
67    /// A `sampling/createMessage` request.
68    CreateMessage(CreateMessageRequest),
69    /// An `elicitation/create` request.
70    Elicitation(ElicitRequest),
71    /// A `roots/list` request.
72    ListRoots(ListRootsRequest),
73}
74
75// Wire-level equality: two `InputRequest`s are equal if they serialize to the
76// same JSON. The wrapped request envelopes do not implement `PartialEq`
77// structurally (they carry `Extensions`).
78impl PartialEq for InputRequest {
79    fn eq(&self, other: &Self) -> bool {
80        match (serde_json::to_value(self), serde_json::to_value(other)) {
81            (Ok(a), Ok(b)) => a == b,
82            _ => false,
83        }
84    }
85}

87/// A map of server-initiated requests that the client must fulfill.
88///
89/// Keys are server-assigned string identifiers; values are request objects
90/// (`ElicitRequest`, `CreateMessageRequest`, or `ListRootsRequest`).
91pub type InputRequests = BTreeMap<String, InputRequest>;
92
93/// A map of client responses to server-initiated requests.
94///
95/// Keys correspond to the keys in the [`InputRequests`] map; values are the
96/// client's result for each request (`ElicitResult`, `CreateMessageResult`,
97/// or `ListRootsResult`), represented as opaque JSON because the
98/// heterogeneous `ClientResult` union does not derive the traits required
99/// for use as a `BTreeMap` value.
100pub type InputResponses = BTreeMap<String, Value>;

102/// Result of a `tools/call` request, including the MRTR intermediate result.
103#[derive(Debug, Clone)]
104#[non_exhaustive]
105pub enum CallToolResponse {
106    /// The server completed the tool call.
107    Complete(CallToolResult),
108    /// The server requires client-side input before the tool call can complete.
109    InputRequired(InputRequiredResult),
110    /// The server materialized a task for this call (SEP-2663 Tasks extension,
111    /// `resultType: "task"`). The client polls `tasks/get` for the result.
112    Task(CreateTaskResult),
113}
114
115impl From<CallToolResult> for CallToolResponse {
116    fn from(result: CallToolResult) -> Self {
117        Self::Complete(result)
118    }
119}
120
121impl From<InputRequiredResult> for CallToolResponse {
122    fn from(result: InputRequiredResult) -> Self {
123        Self::InputRequired(result)
124    }
125}
126
127impl From<CallToolResponse> for ServerResult {
128    fn from(response: CallToolResponse) -> Self {
129        match response {
130            CallToolResponse::Complete(result) => ServerResult::CallToolResult(result),
131            CallToolResponse::InputRequired(result) => ServerResult::InputRequiredResult(result),
132            CallToolResponse::Task(result) => ServerResult::CreateTaskResult(result),
133        }
134    }
135}
136
137impl From<CreateTaskResult> for CallToolResponse {
138    fn from(result: CreateTaskResult) -> Self {
139        Self::Task(result)
140    }
141}

207/// A result indicating that additional input is needed before the request
208/// can be completed.
209///
210/// At least one of [`input_requests`](Self::input_requests) or
211/// [`request_state`](Self::request_state) MUST be present.
212///
213/// Servers MAY send this in response to `tools/call`, `prompts/get`, or
214/// `resources/read`. Servers MUST NOT send this for any other request.
215///
216/// # Examples
217///
218/// ```
219/// use rmcp::model::InputRequiredResult;
220///
221/// let result = InputRequiredResult::from_request_state("opaque-server-state");
222/// assert!(result.input_requests.is_none());
223/// assert_eq!(result.request_state.as_deref(), Some("opaque-server-state"));
224/// ```
225#[derive(Debug, Clone, Serialize)]
226#[serde(rename_all = "camelCase")]
227#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
228#[non_exhaustive]
229pub struct InputRequiredResult {
230    /// Always `"input_required"` for this result type.
231    pub result_type: ResultType,
232
233    /// Server-initiated requests that the client must fulfill before retrying.
234    #[serde(skip_serializing_if = "Option::is_none")]
235    pub input_requests: Option<InputRequests>,
236
237    /// Opaque request state to be echoed back by the client on retry.
238    /// Clients MUST NOT inspect, parse, modify, or make any assumptions
239    /// about the contents.
240    #[serde(skip_serializing_if = "Option::is_none")]
241    pub request_state: Option<String>,
242
243    /// Optional protocol-level metadata.
244    #[serde(rename = "_meta", skip_serializing_if = "Option::is_none")]
245    pub meta: Option<MetaObject>,
246}
247
248/// Custom deserializer that requires `resultType: "input_required"` to prevent
249/// greedy matching in the untagged `ServerResult` enum (which would otherwise
250/// swallow empty objects or unknown shapes).
251impl<'de> Deserialize<'de> for InputRequiredResult {
252    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
253    where
254        D: serde::Deserializer<'de>,
255    {
256        #[derive(Deserialize)]
257        #[serde(rename_all = "camelCase")]
258        struct Helper {
259            result_type: Option<ResultType>,
260            input_requests: Option<InputRequests>,
261            request_state: Option<String>,
262            #[serde(rename = "_meta")]
263            meta: Option<MetaObject>,
264        }
265
266        let helper = Helper::deserialize(deserializer)?;
267
268        match &helper.result_type {
269            Some(rt) if rt.is_input_required() => {}
270            _ => {
271                return Err(serde::de::Error::custom(
272                    "InputRequiredResult requires resultType to be \"input_required\"",
273                ));
274            }
275        }
276
277        if helper.input_requests.is_none() && helper.request_state.is_none() {
278            return Err(serde::de::Error::custom(
279                "InputRequiredResult requires at least one of inputRequests or requestState",
280            ));
281        }
282
283        Ok(InputRequiredResult {
284            result_type: ResultType::INPUT_REQUIRED,
285            input_requests: helper.input_requests,
286            request_state: helper.request_state,
287            meta: helper.meta,
288        })
289    }
290}

292impl InputRequiredResult {
293    /// Creates a new `InputRequiredResult` with both input requests and request state.
294    pub fn new(input_requests: Option<InputRequests>, request_state: Option<String>) -> Self {
295        Self {
296            result_type: ResultType::INPUT_REQUIRED,
297            input_requests,
298            request_state,
299            meta: None,
300        }
301    }
302
303    /// Creates from input requests only.
304    pub fn from_input_requests(input_requests: InputRequests) -> Self {
305        Self::new(Some(input_requests), None)
306    }
307
308    /// Creates from request state only (e.g. for load shedding).
309    pub fn from_request_state(request_state: impl Into<String>) -> Self {
310        Self::new(None, Some(request_state.into()))
311    }
312
313    /// Sets optional metadata.
314    pub fn with_meta(mut self, meta: MetaObject) -> Self {
315        self.meta = Some(meta);
316        self
317    }
318}
